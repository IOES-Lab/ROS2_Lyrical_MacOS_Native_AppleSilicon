  1060	bool AP_Logger_File::io_thread_alive() const
  1061	{
  1062	    if (!hal.scheduler->is_system_initialized()) {
  1063	        // the system has long pauses during initialisation, assume still OK
  1064	        return true;
  1065	    }
  1066	    // if the io thread hasn't had a heartbeat in a while then it is
  1067	#if CONFIG_HAL_BOARD == HAL_BOARD_ESP32
  1068	    uint32_t timeout_ms = 10000;
  1069	#else
  1070	    uint32_t timeout_ms = 5000;
  1071	#endif
  1072	#if CONFIG_HAL_BOARD == HAL_BOARD_SITL && !defined(HAL_BUILD_AP_PERIPH)
  1073	    // the IO thread is working with hardware - writing to a physical
  1074	    // disk.  Unfortunately these hardware devices do not obey our
  1075	    // SITL speedup options, so we allow for it here.
  1076	    SITL::SIM *sitl = AP::sitl();
  1077	    if (sitl != nullptr) {
  1078	        timeout_ms *= sitl->speedup;
  1079	    }
  1080	#endif
  1081	    return (AP_HAL::millis() - _io_timer_heartbeat) < timeout_ms;
  1082	}
